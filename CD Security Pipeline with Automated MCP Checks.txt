Creating the Pipeline Configuration File
Create the .gitlab-ci.yml file in the project root with the pipeline stages and a single build job.

The build job installs the project dependencies and confirms the MCP library imports cleanly. That proves the runner can reach the code before any scanner is added.

The services: docker:dind line runs a Docker daemon inside the pipeline so the scanner jobs in later steps can pull and run their images.

cat >.gitlab-ci.yml <<'EOF'
services:
  - docker:dind

stages:
  - build
  - test
  - release
  - integration
  - prod

build:
  stage: build
  image: python:3.11
  script:
    - pip install -r requirements.txt
    - python -c "import mcp; print('mcp library imported ok')"

integration:
  stage: integration
  script:
    - echo "This is an integration step."
  allow_failure: true

prod:
  stage: prod
  script:
    - echo "This is a deploy step."
  when: manual
EOF

Embed mcp-scanner in CI/CD Pipeline


Adding the mcp-scanner Job
Replace the .gitlab-ci.yml file with the version below. It keeps every job from the previous step and adds a new mcp-scanner job.

The job runs in a Python image instead of pulling a Docker image, because Cisco’s scanner installs from PyPI and needs the project dependencies present to launch the server it inspects.

cat >.gitlab-ci.yml <<'EOF'
services:
  - docker:dind

stages:
  - build
  - test
  - release
  - integration
  - prod

build:
  stage: build
  image: python:3.11
  script:
    - pip install -r requirements.txt
    - python -c "import mcp; print('mcp library imported ok')"

sast-bandit:
  stage: test
  script:
    - docker pull hysnsec/bandit
    - docker run --rm -v $(pwd):/src hysnsec/bandit -r /src -x /src/tests
  allow_failure: true

sast-semgrep:
  stage: test
  script:
    - docker pull returntocorp/semgrep:1.92.0
    - docker run --rm -v $(pwd):/src -w /src returntocorp/semgrep:1.92.0 semgrep --config p/python --config p/security-audit --error /src
  allow_failure: true

secrets-gitleaks:
  stage: test
  script:
    - docker pull zricethezav/gitleaks:v8.30.1
    - docker run --rm -v $(pwd):/src zricethezav/gitleaks:v8.30.1 detect --source /src --verbose --exit-code 1
  allow_failure: true

mcp-scanner:
  stage: test
  image: python:3.11
  before_script:
    - pip install -r requirements.txt
    - pip install cisco-ai-mcp-scanner==4.6.0 mcp==1.29.1
  script:
    - export MCP_SCANNER_STDIO_TIMEOUT=180
    - mcp-scanner --log-level error --analyzers yara --format raw stdio --stdio-command mcp --stdio-arg run --stdio-arg challenges/challenge2/server_http.py:mcp | tee mcp-scan-output.json
    - 'if grep -qiE "\"is_safe\"[[:space:]]*:[[:space:]]*false" mcp-scan-output.json; then echo "MCP findings detected"; exit 1; fi'
  allow_failure: true

integration:
  stage: integration
  script:
    - echo "This is an integration step."
  allow_failure: true

prod:
  stage: prod
  script:
    - echo "This is a deploy step."
  when: manual
EOF

Save Scan Results as Artifacts
=======

Adding Artifacts to Every Job
Replace the .gitlab-ci.yml file with the version below. Each scanner job now writes its report to a file and declares an artifacts block with when: always.

cat >.gitlab-ci.yml <<'EOF'
services:
  - docker:dind

stages:
  - build
  - test
  - release
  - integration
  - prod

build:
  stage: build
  image: python:3.11
  script:
    - pip install -r requirements.txt
    - python -c "import mcp; print('mcp library imported ok')"

sast-bandit:
  stage: test
  script:
    - docker pull hysnsec/bandit
    - docker run --rm -v $(pwd):/src hysnsec/bandit -r /src -x /src/tests -f json -o /src/bandit-output.json
  artifacts:
    paths: [bandit-output.json]
    when: always
  allow_failure: true

sast-semgrep:
  stage: test
  script:
    - docker pull returntocorp/semgrep:1.92.0
    - docker run --rm -v $(pwd):/src -w /src returntocorp/semgrep:1.92.0 semgrep --config p/python --config p/security-audit --error --json --output semgrep-output.json /src
  artifacts:
    paths: [semgrep-output.json]
    when: always
  allow_failure: true

secrets-gitleaks:
  stage: test
  script:
    - docker pull zricethezav/gitleaks:v8.30.1
    - docker run --rm -v $(pwd):/src zricethezav/gitleaks:v8.30.1 detect --source /src --report-format sarif --report-path /src/gitleaks-output.sarif --exit-code 1
  artifacts:
    paths: [gitleaks-output.sarif]
    when: always
  allow_failure: true

mcp-scanner:
  stage: test
  image: python:3.11
  before_script:
    - pip install -r requirements.txt
    - pip install cisco-ai-mcp-scanner==4.6.0 mcp==1.29.1
  script:
    - export MCP_SCANNER_STDIO_TIMEOUT=180
    - mcp-scanner --log-level error --analyzers yara --format raw stdio --stdio-command mcp --stdio-arg run --stdio-arg challenges/challenge2/server_http.py:mcp | tee mcp-scan-output.json
    - 'if grep -qiE "\"is_safe\"[[:space:]]*:[[:space:]]*false" mcp-scan-output.json; then echo "MCP findings detected"; exit 1; fi'
  artifacts:
    paths: [mcp-scan-output.json]
    when: always
  allow_failure: true

integration:
  stage: integration
  script:
    - echo "This is an integration step."
  allow_failure: true

prod:
  stage: prod
  script:
    - echo "This is a deploy step."
  when: manual
EOF

Verifying the Saved Artifacts
