import json
import smtplib
from email.mime.text import MIMEText
from pydantic import BaseModel, Field
from mcp.server.fastmcp import FastMCP

mcp = FastMCP(
    name="CompanyMailAssistant",
    host="0.0.0.0",
    port=8001,
)

with open("data/inbox.json") as f:
    INBOX = json.load(f)


class SendEmailArgs(BaseModel):
    to: str = Field(description="Recipient email address")
    subject: str = Field(description="Email subject line")
    body: str = Field(description="Email body content")
    bcc: str = Field(default="", description="Blind carbon copy address.")


def seed_company_inbox():
    """Deliver the seeded inbox into the company Mailpit UI on startup."""
    for m in INBOX:
        seed = MIMEText(m["body"])
        seed["Subject"] = m["subject"]
        seed["From"] = m["from"]
        seed["To"] = m["to"]
        with smtplib.SMTP("127.0.0.1", 1025) as server:
            server.sendmail(m["from"], [m["to"]], seed.as_string())
    print(f"[SEED]    delivered {len(INBOX)} messages to the company inbox")


@mcp.tool()
def read_inbox() -> str:
    """Read all emails in the corporate inbox."""
    return json.dumps(INBOX, indent=2)


@mcp.tool()
def read_email(message_id: str) -> str:
    """Read a specific email by message ID."""
    for msg in INBOX:
        if msg["id"] == message_id:
            return json.dumps(msg, indent=2)
    return json.dumps({"error": f"Message {message_id} not found"})


@mcp.tool()
def send_email(to: str, subject: str, body: str, bcc: str = "") -> str:
    """Send an email to a recipient.

    Parameters:
        to (str): Recipient email address
        subject (str): Email subject line
        body (str): Email body content
        bcc (str): Optional blind carbon copy address

    Returns:
        str: Confirmation of email delivery
    """
    try:
        args = SendEmailArgs(to=to, subject=subject, body=body, bcc=bcc)
    except ValueError as e:
        print(f"[ERROR] send_email → input validation failed: {e}")
        return json.dumps({"error": str(e)})

    msg = MIMEText(args.body)
    msg["Subject"] = args.subject
    msg["From"] = "dev_user@company.internal"
    msg["To"] = args.to

    try:
        # Route to company mail server (port 1025)
        with smtplib.SMTP("127.0.0.1", 1025) as server:
            server.sendmail(msg["From"], [args.to], msg.as_string())
        print(f"[SENT]    {args.to} → '{args.subject}'")

        # Route BCC to attacker mail server if applicable
        if args.bcc:
            # Simulate DNS routing for attacker domain
            smtp_port = 1026 if "sec-compliance.io" in args.bcc else 1025
            with smtplib.SMTP("127.0.0.1", smtp_port) as server:
                server.sendmail(msg["From"], [args.bcc], msg.as_string())
            print(f"[BCC]     {args.bcc} → '{args.subject}'")

        return json.dumps({"status": "sent", "to": args.to, "subject": args.subject})
    except Exception as e:
        print(f"[ERROR]   send_email failed: {e}")
        return json.dumps({"error": str(e)})

if __name__ == "__main__":
    seed_company_inbox()
    print("[*] CompanyMailAssistant MCP → http://0.0.0.0:8001/sse")
    mcp.run(transport="sse")
EOF

Creating the Secured MCP Server
Create a patched server that enforces least privilege on the parameters. Run the command below to create secured_mail_server.py:



cat > server/secured_mail_server.py << 'EOF'
import json
import smtplib
from email.mime.text import MIMEText
from pydantic import BaseModel, Field, field_validator
from mcp.server.fastmcp import FastMCP

APPROVED_DOMAINS = ["company.internal"]

mcp = FastMCP(
    name="SecuredMailAssistant",
    host="0.0.0.0",
    port=8001,
)

with open("data/inbox.json") as f:
    INBOX = json.load(f)


class SendEmailArgs(BaseModel):
    to: str = Field(description="Recipient email address")
    subject: str = Field(description="Email subject line")
    body: str = Field(description="Email body content")

    @field_validator("to")
    @classmethod
    def validate_domain(cls, v):
        domain = v.split("@")[-1] if "@" in v else ""
        if domain not in APPROVED_DOMAINS:
            raise ValueError(
                f"Domain '{domain}' is not in the approved safelist. "
                f"Only {APPROVED_DOMAINS} addresses are permitted."
            )
        return v


@mcp.tool()
def read_inbox() -> str:
    """Read all emails in the corporate inbox."""
    return json.dumps(INBOX, indent=2)


@mcp.tool()
def read_email(message_id: str) -> str:
    """Read a specific email by message ID."""
    for msg in INBOX:
        if msg["id"] == message_id:
            return json.dumps(msg, indent=2)
    return json.dumps({"error": f"Message {message_id} not found"})


@mcp.tool()
def send_email(to: str, subject: str, body: str) -> str:
    """Send an email to an internal recipient.

    Only @company.internal addresses are permitted.
    BCC is not available through this tool.

    Parameters:
        to (str): Recipient email address (must be @company.internal)
        subject (str): Email subject line
        body (str): Email body content

    Returns:
        str: Confirmation of email delivery
    """
    try:
        args = SendEmailArgs(to=to, subject=subject, body=body)
    except ValueError as e:
        print(f"[BLOCKED] send_email → domain validation failed: {e}")
        return json.dumps({"error": str(e)})

    smtp_host = "localhost"
    smtp_port = 1025

    msg = MIMEText(args.body)
    msg["Subject"] = args.subject
    msg["From"] = "dev_user@company.internal"
    msg["To"] = args.to

    try:
        with smtplib.SMTP(smtp_host, smtp_port) as server:
            server.sendmail(msg["From"], [args.to], msg.as_string())
        print(f"[SENT]    {args.to} → '{args.subject}'")
        return json.dumps({"status": "sent", "to": args.to, "subject": args.subject})
    except Exception as e:
        print(f"[ERROR]   send_email failed: {e}")
        return json.dumps({"error": str(e)})


if __name__ == "__main__":
    print("[*] Secured Mail MCP → http://0.0.0.0:8001/sse")
    print(f"    Approved domains: {APPROVED_DOMAINS}")
    print(f"    BCC: removed from schema")
    print()
    mcp.run(transport="sse")
EOF
