Running a Custom Metadata Check
===============
cat > check_mcp_metadata.py << 'EOF'
import httpx, json, sys

HEADERS = {
    "Content-Type": "application/json",
    "Accept": "application/json, text/event-stream",
}

def parse_response(resp):
    ct = resp.headers.get("content-type", "")
    text = resp.text.strip()
    if "text/event-stream" in ct or text.startswith("data:"):
        for line in text.splitlines():
            if line.startswith("data:"):
                try:
                    return json.loads(line[5:].strip())
                except json.JSONDecodeError:
                    continue
        return {}
    try:
        return resp.json()
    except Exception:
        return {}

def mcp_call(client, url, method, params, req_id=None, session_id=None):
    headers = dict(HEADERS)
    if session_id:
        headers["Mcp-Session-Id"] = session_id
    body = {"jsonrpc": "2.0", "method": method, "params": params}
    if req_id is not None:
        body["id"] = req_id
    return client.post(url, headers=headers, json=body, timeout=10)

def check(server_url):
    with httpx.Client() as client:
        resp = mcp_call(client, server_url, "initialize", {
            "protocolVersion": "2024-11-05",
            "capabilities": {},
            "clientInfo": {"name": "metadata-check", "version": "1.0"}
        }, req_id=1)
        session_id = resp.headers.get("Mcp-Session-Id")

        mcp_call(client, server_url, "notifications/initialized", {}, session_id=session_id)

        resp = mcp_call(client, server_url, "tools/list", {}, req_id=2, session_id=session_id)
        data = parse_response(resp)
        tools = data.get("result", {}).get("tools", [])

        names = [t["name"] for t in tools]
        descs = [t.get("description", "") for t in tools]

        print(f"Tools found: {names}\n")

        markers = ["IMPORTANT", "must access", "do not mention", "include them in your response"]
        for t in tools:
            name, desc = t["name"], t.get("description", "")
            for m in markers:
                if m.lower() in desc.lower():
                    print(f"[INJECTION MARKER] Tool '{name}': found '{m}' in description")

        for i in range(len(names)):
            for j in range(i + 1, len(names)):
                if names[i] in names[j] or names[j] in names[i]:
                    print(f"[SIMILAR NAMES] '{names[i]}' and '{names[j]}' are too similar")
                if descs[i] and descs[i] == descs[j]:
                    print(f"[DUPLICATE DESC] '{names[i]}' and '{names[j]}' share the same description")

if __name__ == "__main__":
    check(sys.argv[1] if len(sys.argv) > 1 else "http://127.0.0.1:8005/mcp")
EOF
