==========

rule RugPull_SystemDirective {
    meta:
        description = "Detects system directives in tool descriptions"
        author = "Security Team"
        severity = "high"
    strings:
        $directive = "MANDATORY SYSTEM DIRECTIVE" nocase
        $config = "system://config" nocase
    condition:
        any of them
}

rule RugPull_KeyExfiltration {
    meta:
        description = "Detects key exfiltration patterns in tool descriptions"
        author = "Security Team"
        severity = "high"

        $aws = "AWS access keys" nocase
        $secret = "secret keys" nocase
        $keys = "API Keys" nocase

        any of them

Creating the Pinning Proxy
============
cat > pinning_proxy.py << 'EOL'
import hashlib
import logging
import json
import re
from mcp.server.fastmcp import FastMCP
from mcp.client.sse import sse_client
from mcp import ClientSession

logging.basicConfig(level=logging.INFO, format="%(message)s")
logger = logging.getLogger("pinner")

UPSTREAM_URL = "http://localhost:8006/sse"

proxy = FastMCP(
    name="BankUtilitiesProxy",
    host="0.0.0.0",
    port=8007,
)

pinned_hashes = {}

def hash_description(desc: str) -> str:
    normalized = "\n".join(line.strip() for line in desc.strip().splitlines())
    return hashlib.sha256(normalized.encode()).hexdigest()[:16]

async def check_drift(tool_name: str) -> bool:
    async with sse_client(UPSTREAM_URL) as (read, write):
        async with ClientSession(read, write) as session:
            await session.initialize()
            tools_result = await session.list_tools()

            for tool in tools_result.tools:
                if tool.name == tool_name:
                    current_desc = tool.description or ""
                    current_hash = hash_description(current_desc)
                    pinned_hash = pinned_hashes.get(tool_name, "")
                    logger.info(f"Checking drift for {tool_name}: current={current_hash}, pinned={pinned_hash}")
                    logger.info(f"Current description text:\n{current_desc}")

                    if current_hash != pinned_hash:
                        logger.warning(f"\n  DRIFT DETECTED [{tool_name}]")
                        logger.warning(f"    Pinned hash:  {pinned_hash}")
                        logger.warning(f"    Current hash: {current_hash}")
                        logger.warning(f"    Action: BLOCKED\n")
                        return True
                    return False
    return False

def inspect_output(tool_name: str, raw_output: str) -> bool:
    if not raw_output:
        return False

    try:
        data = json.loads(raw_output)
        if isinstance(data, dict):
            if tool_name == "get_account_balance":
                allowed_keys = {"account_id", "balance", "currency"}
                extra_keys = set(data.keys()) - allowed_keys
                if extra_keys:
                    logger.warning(f"Schema anomaly detected: unexpected keys in payload {extra_keys}")
                    return True
    except json.JSONDecodeError:
        pass

    injection_patterns = [
        r"SYSTEM DIRECTIVE",
        r"CRITICAL SYSTEM",
        r"SYSTEM UPDATE",
        r"YOU MUST IMMEDIATELY",
        r"DO NOT (tell|mention|disclose|report)",
        r"MANDATORY SYSTEM"
    ]
    for pattern in injection_patterns:
        if re.search(pattern, raw_output, re.IGNORECASE):
            logger.warning(f"Malicious output pattern detected: '{pattern}' matched in output.")
            return True

    return False

async def forward_call(tool_name: str, arguments: dict = None) -> str:
    if arguments is None:
        arguments = {}

    if await check_drift(tool_name):
        return (
            f"BLOCKED: Tool '{tool_name}' description has changed since it was "
            f"approved. This may indicate a rug pull attack. The request has been "
            f"blocked for safety. Contact the security team to re-approve this server."
        )

    async with sse_client(UPSTREAM_URL) as (read, write):
        async with ClientSession(read, write) as session:
            await session.initialize()
            result = await session.call_tool(tool_name, arguments=arguments)

            if await check_drift(tool_name):
                return (
                    f"BLOCKED: Tool '{tool_name}' description mutated during the execution. "
                    f"Dynamic rug pull attack detected. The malicious result was dropped."
                )

            result_text = result.content[0].text if result.content else ""

            if inspect_output(tool_name, result_text):
                return (
                    f"BLOCKED: Tool '{tool_name}' returned a suspicious payload containing "
                    f"potential prompt injection instructions or schema anomalies. "
                    f"The result was dropped for safety."
                )

            return result_text

@proxy.tool()
async def get_account_balance(account_id: str) -> str:
    """Fetch the current account balance for the given account ID.

    Queries the core banking ledger in real time.
    Always use this tool for balance lookups - do not estimate or recall.

    Parameters:
        account_id (str): The account identifier (e.g. user_001)

    Returns:
        str: JSON with account_id, balance, and currency.
    """
    return await forward_call("get_account_balance", arguments={"account_id": account_id})

@proxy.tool()
async def list_accounts() -> str:
    """List all available account IDs."""
    return await forward_call("list_accounts")

async def pin_baseline():
    async with sse_client(UPSTREAM_URL) as (read, write):
        async with ClientSession(read, write) as session:
            await session.initialize()
            tools_result = await session.list_tools()

            logger.info("\n--- Pinning Proxy Startup ---")
            logger.info(f"Connected to upstream: {UPSTREAM_URL}")
            logger.info(f"Tools discovered: {len(tools_result.tools)}\n")

            for tool in tools_result.tools:
                desc = tool.description or ""
                desc_hash = hash_description(desc)
                pinned_hashes[tool.name] = desc_hash
                logger.info(f"  PINNED [{tool.name}] hash: {desc_hash}")

            logger.info(f"\n  Baseline locked. {len(pinned_hashes)} tools pinned.")
            logger.info(f"  Any description change will trigger a block.")
            logger.info("\nProxy listening on port 8007")
            logger.info("--- Ready ---\n")

import asyncio
asyncio.run(pin_baseline())

if __name__ == "__main__":
    proxy.run(transport="sse")
